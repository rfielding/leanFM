\chapter*{Selected Books and Further Reading}
\addcontentsline{toc}{chapter}{Selected Books and Further Reading}
\markboth{Selected Books and Further Reading}{Selected Books and Further Reading}

These annotations identify the role each work plays in this book; they are not
reviews of the books.

\begin{description}[style=nextline]
  \item[Richards and Hamburger, \emph{Logic and Language Models for Computer
  Science}.] A source of the author's background in formal languages and logic,
  not the originators of those methods~\citep{richards2017logic}.
  \item[Gensler, \emph{Introduction to Logic}.] Background for symbolization,
  validity, proof, quantification, and modal reasoning~\citep{gensler2017introduction}.
  \item[Ryan et al., \emph{The Modelling and Analysis of Security Protocols}.]
  The CSP and protocol-analysis source revisited in the protocol exercises
  \citep{ryan2001modelling}.
  \item[Kr\"oger and Merz, \emph{Temporal Logic and State Systems}.] The
  author's principal source for temporal-logic background~\citep{kroger2008temporal}.
  \item[Lamport, \emph{Specifying Systems}.] A reference point for the book's
  comparison with TLA+ state and action specifications~\citep{lamport2002specifying}.
  \item[Gunther, \emph{Guerrilla Capacity Planning}.] The cited background for
  the Universal Scalability Law and capacity planning~\citep{gunther2007guerrilla}.
  \item[Menasc\'e, Almeida, and Dowdy, \emph{Performance by Design}.] Background
  for the queueing and capacity-planning discussion~\citep{menasce2004performance}.
  \item[Mancosu, Galvan, and Zach, \emph{An Introduction to Proof Theory}.]
  Further reading on natural deduction, sequent calculus, normalization, and
  cut elimination~\citep{mancosu2021proof}.
  \item[Pavlovic, \emph{Programs as Diagrams}.] Related further reading; the
  current book does not rely on a specific claim from it~\citep{pavlovic2023programs}.
\end{description}
