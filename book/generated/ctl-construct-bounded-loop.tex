\begin{lstlisting}[language={},basicstyle=\ttfamily\footnotesize,breaklines=true]
protocol Retry (A, B) {
  loop Try at most 3 {
    q: Ask from A to B;
    choice {
      Ok from B to A cause q;
    } or {
      Busy from B to A cause q;
      continue Try;
    }
  } otherwise {
    GiveUp from A to B;
  }
}
\end{lstlisting}
