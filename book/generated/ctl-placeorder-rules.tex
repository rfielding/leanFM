  ¬(CreateOrder ∨ Quote ∨ SoldOut ∨ Pay ∨ Charge ∨ Charged ∨ Declined ∨ Receipt ∨ PaymentDeclined) ∧ ¬stop

  always ((¬(CreateOrder ∨ Quote ∨ SoldOut ∨ Pay ∨ Charge ∨ Charged ∨ Declined ∨ Receipt ∨ PaymentDeclined) ∧ ¬stop) → (someNext (CreateOrder ∧ ¬stop) ∧ allNext (CreateOrder ∧ ¬stop)))

  always ((Charged ∧ ¬stop) → (someNext (Receipt ∧ stop) ∧ allNext (Receipt ∧ stop)))

typestate 7: Declined
  always ((Declined ∧ ¬stop) → (someNext (PaymentDeclined ∧ stop) ∧ allNext (PaymentDeclined ∧ stop)))

typestate 8: Receipt (may end)
  always ((Receipt ∧ stop) → (someNext (Receipt ∧ stop) ∧ allNext (Receipt ∧ stop)))

typestate 9: PaymentDeclined (may end)
  always ((PaymentDeclined ∧ stop) → (someNext (PaymentDeclined ∧ stop) ∧ allNext (PaymentDeclined ∧ stop)))
